cmake_minimum_required(VERSION 3.16)
project(task_scheduler VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(TASK_SCHEDULER_ASAN_UBSAN "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
option(TASK_SCHEDULER_TSAN "Enable ThreadSanitizer (separate build)" OFF)
option(TASK_SCHEDULER_COVERAGE "Enable GCC/Clang coverage instrumentation" OFF)

if(TASK_SCHEDULER_ASAN_UBSAN AND TASK_SCHEDULER_TSAN)
    message(FATAL_ERROR "ASan and TSan require separate builds")
endif()

add_library(scheduler_options INTERFACE)
if(MSVC)
    target_compile_options(scheduler_options INTERFACE /W4 /permissive-)
else()
    target_compile_options(scheduler_options INTERFACE -Wall -Wextra -Wpedantic)
endif()

if(TASK_SCHEDULER_ASAN_UBSAN OR TASK_SCHEDULER_TSAN)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Sanitizer options require GCC or Clang")
    endif()
    if(TASK_SCHEDULER_ASAN_UBSAN)
        set(SANITIZERS -fsanitize=address,undefined)
    else()
        set(SANITIZERS -fsanitize=thread)
    endif()
    target_compile_options(scheduler_options INTERFACE ${SANITIZERS} -fno-omit-frame-pointer)
    target_link_options(scheduler_options INTERFACE ${SANITIZERS})
endif()

if(TASK_SCHEDULER_COVERAGE)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Coverage options require GCC or Clang")
    endif()
    target_compile_options(scheduler_options INTERFACE --coverage -fprofile-update=atomic -O0 -g)
    target_link_options(scheduler_options INTERFACE --coverage)
endif()

find_package(Threads REQUIRED)
add_library(task_scheduler src/thread_pool.cpp)
target_include_directories(task_scheduler PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/src)
target_compile_features(task_scheduler PUBLIC cxx_std_17)
target_link_libraries(task_scheduler PUBLIC Threads::Threads)
target_compile_options(task_scheduler PRIVATE $<TARGET_PROPERTY:scheduler_options,INTERFACE_COMPILE_OPTIONS>)
target_link_options(task_scheduler PRIVATE $<TARGET_PROPERTY:scheduler_options,INTERFACE_LINK_OPTIONS>)

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
    add_executable(scheduler_example src/examples/basic.cpp)
    target_link_libraries(scheduler_example PRIVATE task_scheduler scheduler_options)

    add_executable(scheduler_benchmark src/benchmarks/throughput.cpp)
    target_link_libraries(scheduler_benchmark PRIVATE task_scheduler scheduler_options)

    include(CTest)
    if(BUILD_TESTING)
        add_executable(scheduler_tests src/tests/scheduler_tests.cpp)
        target_link_libraries(scheduler_tests PRIVATE task_scheduler scheduler_options)
        add_test(NAME scheduler_tests COMMAND scheduler_tests)
        set_tests_properties(scheduler_tests PROPERTIES TIMEOUT 120)
        if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            add_executable(scheduler_failure_tests src/tests/failure_tests.cpp)
            target_link_libraries(scheduler_failure_tests PRIVATE task_scheduler scheduler_options ${CMAKE_DL_LIBS})
            if(TASK_SCHEDULER_COVERAGE AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
                target_compile_definitions(scheduler_failure_tests PRIVATE TASK_SCHEDULER_GCOV=1)
            endif()
            add_test(NAME scheduler_failure_tests COMMAND scheduler_failure_tests)
            set_tests_properties(scheduler_failure_tests PROPERTIES TIMEOUT 30)
        endif()
    endif()
endif()

include(GNUInstallDirs)
install(TARGETS task_scheduler)
install(DIRECTORY src/task_scheduler DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}
    FILES_MATCHING PATTERN "*.hpp")
